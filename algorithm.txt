------BIT treeについて----
BITTreeのインデックス　S = {0,1}1{0}　は {0,1}0{0} ~~ {0,1}1{0}　までの合計をもつ
つまり例えば1100は番号　1000 ~~ 1100　までの和をもつ
ある数字110100までの和は
SUM += 
110100 = 110000 ~~ 110100
110000 = 100000 ~~ 110000
100000 = 000000 ~~ 100000
0 = N/A

で計算できる
class BIT:
  def __init__(self,n):
    self.data = [0]*(n+1)
  
  def sum(self,i):
    r = 0
    while i > 0:
      r += self.data[i]
      i -= i&(-i)
    return r
  
  def add(self,i,x):
    while i <= n:
      self.data[i] += x
      i += i&(-i)

------SegmentTree------
class SegmentTree:
  def __init__(self,n):
    self.size = 1
    while self.size < n:
      self.size *= 2
    self.data = [0]*(2*self.size)
  
  def update(self,u,x):
    u += self.size
    self.data[u] = x
    while u > 1:
      u //= 2
      self.data[u] = self.data[u*2]+self.data[u*2+1]
  
  def query(self,l,r,a,b,u):
    if b <= l or r <= a:
      return 0
    if l<=a and b <= r:
      return self.data[u]
    
    m = (a+b)//2
    ar = self.query(l,r,a,m,u*2)
    al = self.query(l,r,m,b,u*2+1)
    
    return ar+al
  
  def range_query(self,l,r):
    return self.query(l,r,0,self.size,1)

-----UnionFind------
class UnionFind:
  def __init__(self,n):
    self.parent = [i for i in range(n)]
    
  def find(self,i):
    if self.parent[i] == i:
      return i
    self.parent[i] = self.find(self.parent[i])
    return self.parent[i]
  
  def union(self,i,j):
    root_i = self.find(i)
    root_j = self.find(j)
    
    if root_i == root_j:
      return
    
    self.parent[root_i] = root_j
  
  def same(self,u,v):
    return self.find(u) == self.find(v)
